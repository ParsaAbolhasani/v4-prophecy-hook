pragma solidity ^0.8.24;

import {Test, console} from "forge-std/Test.sol";
import {SettlementMath} from "../src/libraries/SettlementMath.sol";

contract SettlementMathTest is Test {
    // ─── computePayout ──────────────────────────────────────────────

    function test_ComputePayout_BaseCase() public {
        uint256 collateral = 1000e18;
        uint256 feeBps = 200; // 2%

        uint256 payout = SettlementMath.computePayout(collateral, feeBps);

        // 1000 - (1000 * 0.02) = 980
        assertEq(payout, 980e18);
    }

    function test_ComputePayout_MaxFee() public {
        uint256 collateral = 1000e18;
        uint256 feeBps = 500; // 5%

        uint256 payout = SettlementMath.computePayout(collateral, feeBps);

        // 1000 - (1000 * 0.05) = 950
        assertEq(payout, 950e18);
    }

    function test_ComputePayout_RevertsOnZeroCollateral() public {
        vm.expectRevert(SettlementMath.InvalidCollateral.selector);
        SettlementMath.computePayout(0, 200);
    }

    function test_ComputePayout_RevertsOnInvalidFee() public {
        vm.expectRevert(SettlementMath.InvalidFee.selector);
        SettlementMath.computePayout(1000e18, 600); // > 5%
    }

    // ─── computeDynamicFee ──────────────────────────────────────────

    function test_ComputeDynamicFee_BaseCase() public {
        uint256 fee = SettlementMath.computeDynamicFee(200, 100);
        assertEq(fee, 300); // 2% + 1% = 3%
    }

    function test_ComputeDynamicFee_CapsAtMax() public {
        uint256 fee = SettlementMath.computeDynamicFee(200, 500); // 2% + 5%
        assertEq(fee, 500); // capped at 5%
    }

    function test_ComputeDynamicFee_ZeroVolatility() public {
        uint256 fee = SettlementMath.computeDynamicFee(200, 0);
        assertEq(fee, 200);
    }

    // ─── computeShares ──────────────────────────────────────────────

    function test_ComputeShares_FirstDeposit() public {
        uint256 shares = SettlementMath.computeShares(1000e18, 0, 0);
        assertEq(shares, 1000e18); // 1:1 for first deposit
    }

    function test_ComputeShares_SubsequentDeposit() public {
        // 1000 deposited, 1000 shares minted
        // Now deposit 500 more → should get 500 shares
        uint256 shares = SettlementMath.computeShares(500e18, 1000e18, 1000e18);
        assertEq(shares, 500e18);
    }

    function test_ComputeShares_ProportionalAfterYield() public {
        // 1000 deposited, 1000 shares. Yield accrued: 1000 deposits, 1100 total
        // Deposit 500 → should get 500 * 1000 / 1100 = 454.54...
        uint256 shares = SettlementMath.computeShares(500e18, 1100e18, 1000e18);
        assertApproxEqRel(shares, 454.54e18, 0.01e18); // 1% tolerance
    }

    function test_ComputeShares_RevertsOnZeroAmount() public {
        vm.expectRevert(SettlementMath.InvalidCollateral.selector);
        SettlementMath.computeShares(0, 1000e18, 1000e18);
    }

    // ─── computeWithdrawAmount ──────────────────────────────────────

    function test_ComputeWithdrawAmount_BaseCase() public {
        uint256 amount = SettlementMath.computeWithdrawAmount(
            500e18,
            1000e18,
            1000e18
        );
        assertEq(amount, 500e18);
    }

    function test_ComputeWithdrawAmount_RevertsOnZeroShares() public {
        vm.expectRevert(SettlementMath.InvalidCollateral.selector);
        SettlementMath.computeWithdrawAmount(0, 1000e18, 1000e18);
    }

    // ─── computeVolatilityBps ───────────────────────────────────────

    function test_ComputeVolatilityBps_NoChange() public {
        uint256 vol = SettlementMath.computeVolatilityBps(100, 100);
        assertEq(vol, 0);
    }

    function test_ComputeVolatilityBps_1Percent() public {
        // Price goes from 100 to 101 → 1% = 100 bps
        uint256 vol = SettlementMath.computeVolatilityBps(101, 100);
        assertEq(vol, 100);
    }

    function test_ComputeVolatilityBps_10Percent() public {
        // Price goes from 100 to 110 → 10% = 1000 bps
        uint256 vol = SettlementMath.computeVolatilityBps(110, 100);
        assertEq(vol, 1000);
    }

    function test_ComputeVolatilityBps_HandlesZeroPrevious() public {
        uint256 vol = SettlementMath.computeVolatilityBps(100, 0);
        assertEq(vol, 0);
    }

    // ─── absInt128 ──────────────────────────────────────────────────

    function test_AbsInt128_Positive() public {
        assertEq(SettlementMath.absInt128(100), 100);
    }

    function test_AbsInt128_Negative() public {
        assertEq(SettlementMath.absInt128(-100), 100);
    }

    function test_AbsInt128_Zero() public {
        assertEq(SettlementMath.absInt128(0), 0);
    }

    // ─── Fuzz Tests ─────────────────────────────────────────────────

    function testFuzz_ComputePayout_NeverExceedsCollateral(
        uint256 collateral,
        uint256 feeBps
    ) public {
        collateral = bound(collateral, 1, type(uint128).max);
        feeBps = bound(feeBps, 0, 500);

        uint256 payout = SettlementMath.computePayout(collateral, feeBps);

        assertLe(payout, collateral);
        assertGe(payout, (collateral * 95) / 100); // At least 95%
    }

    function testFuzz_ComputeShares_Proportional(
        uint256 amount,
        uint256 totalDeposits,
        uint256 totalShares
    ) public {
        amount = bound(amount, 1, type(uint128).max);
        totalDeposits = bound(totalDeposits, 1, type(uint128).max);
        totalShares = bound(totalShares, 1, type(uint128).max);

        uint256 shares = SettlementMath.computeShares(
            amount,
            totalDeposits,
            totalShares
        );

        // Shares should be proportional
        assertLe(shares, (amount * totalShares) / totalDeposits + 1);
    }
}
