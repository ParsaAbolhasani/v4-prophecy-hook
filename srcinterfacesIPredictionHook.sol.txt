pragma solidity ^0.8.24;

import {PoolKey} from "v4-core/types/PoolKey.sol";
import {PoolId} from "v4-core/types/PoolId.sol";

/// @title IPredictionHook
/// @notice Interface for the Prophecy prediction hook
interface IPredictionHook {
    // ─── Structs ────────────────────────────────────────────────────

    struct Round {
        uint64 startTime;
        uint64 endTime;
        uint256 lockedCollateral;
        uint256 dynamicFeeBps;
        address initiator;
        bool settled;
    }

    // ─── Events ─────────────────────────────────────────────────────

    event RoundOpened(
        PoolId indexed poolId,
        address indexed initiator,
        uint256 collateral,
        uint256 dynamicFeeBps
    );

    event RoundSettled(
        PoolId indexed poolId,
        address indexed winner,
        uint256 payout,
        bool usedJIT
    );

    event DynamicFeeAdjusted(
        PoolId indexed poolId,
        uint256 newFeeBps,
        int256 volatility
    );

    // ─── Errors ─────────────────────────────────────────────────────

    error StaleOraclePrice();
    error LowConfidenceOracle();
    error RoundAlreadyActive();
    error RoundNotActive();
    error UnauthorizedSettlement();
    error InvalidDuration();
    error InvalidCollateral();

    // ─── Functions ──────────────────────────────────────────────────

    function openRound(
        PoolKey calldata key,
        uint64 duration,
        uint256 collateral
    ) external;

    function getRound(PoolId poolId) external view returns (Round memory);

    function isRoundActive(PoolId poolId) external view returns (bool);
}
