pragma solidity ^0.8.24;

import {Test, console} from "forge-std/Test.sol";
import {PredictionHook} from "../src/PredictionHook.sol";
import {SingleSidedVault} from "../src/SingleSidedVault.sol";
import {JITSettlementEngine} from "../src/JITSettlementEngine.sol";
import {MockPyth} from "../src/mocks/MockPyth.sol";
import {MockERC20} from "./mocks/MockERC20.sol";
import {PoolKey} from "v4-core/types/PoolKey.sol";
import {PoolId, PoolIdLibrary} from "v4-core/types/PoolId.sol";
import {Currency} from "v4-core/types/Currency.sol";
import {IHooks} from "v4-core/interfaces/IHooks.sol";
import {IPoolManager} from "v4-core/interfaces/IPoolManager.sol";

/// @title PredictionHookInvariantTest
/// @notice Stateful invariant tests for Prophecy Hook
contract PredictionHookInvariantTest is Test {
    using PoolIdLibrary for PoolKey;

    PredictionHook public hook;
    SingleSidedVault public vault;
    JITSettlementEngine public engine;
    MockPyth public pyth;
    MockERC20 public collateralToken;

    address public alice = address(0xA11CE);
    address public bob = address(0xB0B);

    PoolKey public poolKey;
    bytes32 public constant PRICE_FEED_ID = bytes32(uint256(1));

    function setUp() public {
        // Deploy mocks
        pyth = new MockPyth();
        pyth.setPriceSimple(PRICE_FEED_ID, 1e8); // 1.0

        collateralToken = new MockERC20("Collateral", "COL", 18);

        // Deploy vault (placeholder hook, will update)
        vault = new SingleSidedVault(address(0), address(0));

        // Deploy engine
        engine = new JITSettlementEngine(
            address(0xPM), // placeholder pool manager
            address(vault)
        );

        // Deploy hook with pyth mock
        // In real invariant test, use vm.etch to set code at hook address
        // For simplicity, we'll use a real deployment

        poolKey = PoolKey({
            currency0: Currency.wrap(address(collateralToken)),
            currency1: Currency.wrap(address(0x2)),
            fee: 3000,
            tickSpacing: 60,
            hooks: IHooks(address(hook))
        });

        // Fund actors
        collateralToken.mint(alice, 100_000e18);
        collateralToken.mint(bob, 100_000e18);

        // Setup actors for invariant
        targetContract(address(this));
    }

    // ─── Invariant Functions ────────────────────────────────────────

    /// @notice Vault balance should never be negative (always >= 0)
    function invariant_VaultBalanceNonNegative() public view {
        uint256 balance = vault.balanceOf(address(collateralToken));
        assertGe(balance, 0, "vault balance underflow");
    }

    /// @notice Total shares should equal sum of individual shares
    function invariant_TotalSharesConsistent() public view {
        uint256 aliceShares = vault.sharesOf(address(collateralToken), alice);
        uint256 bobShares = vault.sharesOf(address(collateralToken), bob);

        // Note: This requires a getter for total shares
        // For now, just verify individual shares are non-negative
        assertGe(aliceShares, 0);
        assertGe(bobShares, 0);
    }

    /// @notice Hook permissions should never change
    function invariant_HookPermissionsConstant() public view {
        // Hook permissions are set in constructor and should be immutable
        // This is a sanity check that they return valid values
        // In practice, permissions are pure and cannot change
        assertTrue(true);
    }

    // ─── Handler Functions ──────────────────────────────────────────

    /// @notice Handler: Alice deposits
    function depositAlice(uint256 amount) external {
        amount = bound(amount, 1e18, 10_000e18);

        vm.startPrank(alice);
        collateralToken.approve(address(vault), amount);
        vault.deposit(address(collateralToken), amount);
        vm.stopPrank();
    }

    /// @notice Handler: Bob deposits
    function depositBob(uint256 amount) external {
        amount = bound(amount, 1e18, 10_000e18);

        vm.startPrank(bob);
        collateralToken.approve(address(vault), amount);
        vault.deposit(address(collateralToken), amount);
        vm.stopPrank();
    }

    /// @notice Handler: Alice withdraws
    function withdrawAlice(uint256 shares) external {
        uint256 balance = vault.sharesOf(address(collateralToken), alice);
        if (balance == 0) return;

        shares = bound(shares, 1, balance);

        vm.prank(alice);
        vault.withdraw(address(collateralToken), shares);
    }
}
